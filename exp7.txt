import java.util.Scanner;

public class StringHandlingDemo {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter a string: ");
        String str = sc.nextLine();

        System.out.println("\n--- String Operations ---");
        System.out.println("Original String: " + str);

        // Length of string
        System.out.println("Length of string: " + str.length());

        // Uppercase and Lowercase
        System.out.println("Uppercase: " + str.toUpperCase());
        System.out.println("Lowercase: " + str.toLowerCase());

        // Substring (with a check to prevent StringIndexOutOfBoundsException)
        if (str.length() >= 5) {
            System.out.println("Substring (0 to 5): " + str.substring(0, 5));
        } else {
            System.out.println("Substring (0 to 5): [String too short]");
        }

        // Replace
        System.out.println("Replace 'a' with '@': " + str.replace('a', '@'));

        // String comparison
        System.out.print("\nEnter another string to compare: ");
        String str2 = sc.nextLine();

        if (str.equals(str2)) {
            System.out.println("Result: Both strings are equal.");
        } else {
            System.out.println("Result: Both strings are not equal.");
        }

        sc.close();
    }
}
